```python
    token: str = Query(...)

    """对话WebSocket连接"""
    
    # 1. 验证用户身份

    )
    if not conversation:
        await websocket.close(code=1008, reason="对话不存在")
        return
    
    await websocket.accept()
    
    try:
        while True:
            # 接收客户端消息
            data = await websocket.receive_json()
            
            if data["type"] == "ping":
                await websocket.send_json({"type": "pong"})
            
            elif data["type"] == "message":
                # 处理消息并发送进度更新
                await handle_websocket_message(
                    websocket, conversation_id, user.id, data["message"]
                )
                
    except WebSocketDisconnect:
        logger.info(f"WebSocket disconnected for conversation {conversation_id}")
```

```python
# app/services/cache_optimization_service.py

class CacheOptimizationService:
    """智能缓存优化服务"""
    
    def __init__(self):
        self.redis_client = Redis()
        self.cache_stats = CacheStatsCollector()
    
    async def get_or_compute_analysis(
        self, 
        cache_key: str,
        compute_func: Callable,
        ttl: int = 3600
    ):
        """获取或计算分析结果"""
        
        # 1. 尝试从缓存获取
        cached_result = await self.redis_client.get(cache_key)
        if cached_result:
            self.cache_stats.record_hit(cache_key)
            return json.loads(cached_result)
        
        # 2. 计算新结果
        result = await compute_func()
        
        # 3. 存储到缓存
        await self.redis_client.setex(
            cache_key, 
            ttl, 
            json.dumps(result, default=str)
        )
        
        self.cache_stats.record_miss(cache_key)
        return result
    
    def generate_cache_key(
        self, 
        intent: str, 
        message: str, 
        expert_type: str
    ) -> str:
        """生成智能缓存键"""
        
        # 提取关键信息
        keywords = self._extract_keywords(message)
        
        # 生成哈希键
        key_data = {
            'intent': intent,
            'keywords': sorted(keywords),
            'expert': expert_type,
            'version': 'v3.0'
        }
        
        key_string = json.dumps(key_data, sort_keys=True)
        return f"analysis:{hashlib.md5(key_string.encode()).hexdigest()}"
```

```python
# app/services/concurrent_analysis_service.py

class ConcurrentAnalysisService:
    """并发分析处理服务"""
    
    def __init__(self):
        self.semaphore = asyncio.Semaphore(10)  # 限制并发数
        self.task_queue = asyncio.Queue()
        self.worker_pool = []
    
    async def process_analysis_request(
        self, 
        request: AnalysisRequest
    ) -> AnalysisResults:
        """处理分析请求"""
        
        async with self.semaphore:
            # 并行执行多个分析任务
            tasks = [
                self._run_trends_analysis(request),
                self._run_pmf_analysis(request),
                self._run_insights_analysis(request)
            ]
            
            # 等待所有任务完成
            results = await asyncio.gather(*tasks, return_exceptions=True)
            
            # 处理异常和整合结果
            return self._integrate_results(results)
    
    async def _run_trends_analysis(self, request: AnalysisRequest):
        """运行趋势分析"""
        try:
            return await trends_service.analyze_comprehensive(request.keyword)
        except Exception as e:
            logger.error(f"趋势分析失败: {e}")
            return {"error": str(e), "type": "trends_analysis"}
```
