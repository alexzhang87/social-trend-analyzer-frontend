```python
class HuggingFaceDataLoader:
    def __init__(self):
        self.datasets_config = HUGGINGFACE_DATASETS
        
    async def load_dataset(self, dataset_key: str) -> Dict:
        """加载Hugging Face数据集"""
        config = self.datasets_config[dataset_key]
        
        # 使用datasets库加载
        from datasets import load_dataset
        dataset = load_dataset(
            config["dataset_name"],
            config.get("subset"),
            split="train"
        )
        
        # 数据预处理
        processed_data = self.preprocess_conversations(dataset)
        
        return {
            "raw_data": dataset,
            "processed_data": processed_data,
            "metadata": config
        }
    
    def preprocess_conversations(self, dataset) -> List[Dict]:
        """预处理对话数据"""
        processed = []
        
        for item in dataset:
            # 提取对话对
            if "conversation" in item:
                conversation = item["conversation"]
            elif "messages" in item:
                conversation = item["messages"]
            else:
                continue
                
            # 转换为训练格式
            training_pairs = self.extract_qa_pairs(conversation)
            processed.extend(training_pairs)
            
        return processed
    
    def extract_qa_pairs(self, conversation: List[Dict]) -> List[Dict]:
        """提取问答对"""
        pairs = []
        
        for i in range(0, len(conversation)-1, 2):
            if i+1 < len(conversation):
                question = conversation[i].get("content", "")
                answer = conversation[i+1].get("content", "")
                
                if len(question) > 20 and len(answer) > 20:
                    pairs.append({
                        "input": question,
                        "output": answer,
                        "source": "huggingface",
                        "quality_score": self.assess_quality(question, answer)
                    })
                    
        return pairs
```

```python
class StackOverflowDataLoader:
    def __init__(self):
        self.api_base = "https://api.stackexchange.com/2.3"
        self.site = "stackoverflow"
        self.rate_limit = 300  # requests per second
        
    async def download_data_dump(self) -> str:
        """下载Stack Overflow数据转储"""
        # 使用BitTorrent下载完整数据集
        torrent_url = "https://archive.org/download/stackexchange"
        
        # 实现BitTorrent下载逻辑
        import libtorrent as lt
        
        session = lt.session()
        info = lt.torrent_info(torrent_url)
        handle = session.add_torrent({
            'ti': info,
            'save_path': './data/stackoverflow/'
        })
        
        # 等待下载完成
        while not handle.is_seed():
            await asyncio.sleep(1)
            
        return "./data/stackoverflow/Posts.xml"
    
    def parse_xml_dump(self, xml_file: str) -> List[Dict]:
        """解析XML数据转储"""
        import xml.etree.ElementTree as ET
        
        tree = ET.parse(xml_file)
        root = tree.getroot()
        
        qa_pairs = []
        questions = {}
        
        # 第一遍：收集所有问题
        for post in root.findall('.//row'):
            post_type = post.get('PostTypeId')
            if post_type == '1':  # Question
                questions[post.get('Id')] = {
                    'title': post.get('Title'),
                    'body': post.get('Body'),
                    'tags': post.get('Tags'),
                    'score': int(post.get('Score', 0))
                }
        
        # 第二遍：匹配答案
        for post in root.findall('.//row'):
            post_type = post.get('PostTypeId')
            if post_type == '2':  # Answer
                parent_id = post.get('ParentId')
                if parent_id in questions:
                    question = questions[parent_id]
                    answer = {
                        'body': post.get('Body'),
                        'score': int(post.get('Score', 0)),
                        'accepted': post.get('Id') == questions[parent_id].get('AcceptedAnswerId')
                    }
                    
                    # 只保留高质量问答对
                    if question['score'] >= 5 and answer['score'] >= 3:
                        qa_pairs.append({
                            'input': f"{question['title']}\n{question['body']}",
                            'output': answer['body'],
                            'tags': question['tags'],
                            'quality_score': self.calculate_quality_score(question, answer),
                            'source': 'stackoverflow'
                        })
        
        return qa_pairs
    
    def filter_relevant_content(self, qa_pairs: List[Dict]) -> List[Dict]:
        """过滤与创业技术相关的内容"""
        relevant_tags = {
            'startup', 'business', 'saas', 'mvp', 'product-management',
            'analytics', 'metrics', 'growth', 'marketing', 'api-design',
            'scalability', 'architecture', 'database-design', 'security'
        }
        
        filtered_pairs = []
        for pair in qa_pairs:
            tags = set(pair['tags'].lower().split())
            if tags.intersection(relevant_tags):
                # 转换技术问题为创业技术咨询
                adapted_pair = self.adapt_to_startup_context(pair)
                filtered_pairs.append(adapted_pair)
                
        return filtered_pairs
    
    def adapt_to_startup_context(self, qa_pair: Dict) -> Dict:
        """将技术问答适配为创业技术咨询"""
        input_text = qa_pair['input']
        output_text = qa_pair['output']
        
        # 添加创业背景上下文
        startup_context = "As a startup founder, "
        if not input_text.startswith(startup_context):
            input_text = startup_context + input_text.lower()
        
        # 调整回答语调为咨询式
        consulting_prefix = "For your startup, I'd recommend: "
        if not output_text.startswith(consulting_prefix):
            output_text = consulting_prefix + output_text
        
        return {
            'input': input_text,
            'output': output_text,
            'domain': 'startup_tech_consulting',
            'source': 'stackoverflow_adapted',
            'quality_score': qa_pair['quality_score']
        }
```

```python
class UnifiedDataProcessor:
    def __init__(self):
        self.quality_threshold = 0.8
        self.min_length = 50
        self.max_length = 2048
        self.language_detector = self.init_language_detector()
        
    async def process_all_sources(self) -> Dict:
        """处理所有数据源"""
        all_data = []
        
        # 1. 处理Hugging Face数据
        hf_loader = HuggingFaceDataLoader()
        for dataset_key in HUGGINGFACE_DATASETS.keys():
            data = await hf_loader.load_dataset(dataset_key)
            all_data.extend(data['processed_data'])
        
        # 2. 处理Stack Overflow数据
        so_loader = StackOverflowDataLoader()
        xml_file = await so_loader.download_data_dump()
        so_data = so_loader.parse_xml_dump(xml_file)
        filtered_so_data = so_loader.filter_relevant_content(so_data)
        all_data.extend(filtered_so_data)
        
        # 3. 处理SCORE数据
        score_collector = SCOREDataCollector()
        score_data = await score_collector.collect_public_resources()
        all_data.extend(score_data)
        
        # 4. 统一质量控制
        cleaned_data = self.apply_quality_control(all_data)
        
        # 5. 数据去重
        deduplicated_data = self.remove_duplicates(cleaned_data)
        
        # 6. 格式标准化
        standardized_data = self.standardize_format(deduplicated_data)
        
        return {
            "total_samples": len(standardized_data),
            "data": standardized_data,
            "quality_report": self.generate_quality_report(standardized_data)
        }
    
    def apply_quality_control(self, raw_data: List[Dict]) -> List[Dict]:
        """应用质量控制"""
        cleaned_data = []
        
        for item in raw_data:
            # 1. 语言检测
            if not self.is_english(item['input']) or not self.is_english(item['output']):
                continue
            
            # 2. 长度过滤
            if not (self.min_length <= len(item['input']) <= self.max_length):
                continue
            if not (self.min_length <= len(item['output']) <= self.max_length):
                continue
            
            # 3. 质量评分
            quality_score = self.calculate_quality_score(item)
            if quality_score < self.quality_threshold:
                continue
            
            # 4. 内容安全检查
            if not self.is_safe_content(item):
                continue
            
            item['quality_score'] = quality_score
            cleaned_data.append(item)
        
        return cleaned_data
    
    def standardize_format(self, data: List[Dict]) -> List[Dict]:
        """标准化数据格式"""
        standardized = []
        
        for item in data:
            standardized_item = {

                "quality_score": item['quality_score'],
```

```python
class ModelTrainingIntegration:
    def __init__(self):
        self.model_name = "microsoft/DialoGPT-medium"  # 基础模型
        self.training_config = {
            "learning_rate": 2e-4,
            "batch_size": 16,
            "num_epochs": 3,
            "max_length": 1024,
            "lora_r": 16,
            "lora_alpha": 32,
            "lora_dropout": 0.05
        }
    
    async def train_with_free_data(self, processed_data: List[Dict]) -> Dict:
        """使用免费数据训练模型"""
        
        # 1. 数据集划分
        train_data, val_data, test_data = self.split_dataset(processed_data)
        
        # 2. 创建训练数据集
        train_dataset = self.create_training_dataset(train_data)
        val_dataset = self.create_training_dataset(val_data)
        
        # 3. 配置LoRA微调
        from peft import LoraConfig, get_peft_model
        
        lora_config = LoraConfig(
            r=self.training_config["lora_r"],
            lora_alpha=self.training_config["lora_alpha"],
            lora_dropout=self.training_config["lora_dropout"],
            target_modules=["q_proj", "v_proj", "k_proj", "o_proj"],
            bias="none",
            task_type="CAUSAL_LM"
        )
        
        # 4. 加载基础模型
        from transformers import AutoModelForCausalLM, AutoTokenizer
        
        model = AutoModelForCausalLM.from_pretrained(self.model_name)
        tokenizer = AutoTokenizer.from_pretrained(self.model_name)
        
        # 5. 应用LoRA
        model = get_peft_model(model, lora_config)
        
        # 6. 训练配置
        from transformers import TrainingArguments, Trainer
        
        training_args = TrainingArguments(
            output_dir="./models/ideaeden-consultant",
            num_train_epochs=self.training_config["num_epochs"],
            per_device_train_batch_size=self.training_config["batch_size"],
            learning_rate=self.training_config["learning_rate"],
            logging_steps=100,
            save_steps=500,
            evaluation_strategy="steps",
            eval_steps=500,
            load_best_model_at_end=True
        )
        
        # 7. 创建训练器
        trainer = Trainer(
            model=model,
            args=training_args,
            train_dataset=train_dataset,
            eval_dataset=val_dataset,
            tokenizer=tokenizer
        )
        
        # 8. 开始训练
        training_result = trainer.train()
        
        # 9. 保存模型
        model.save_pretrained("./models/ideaeden-consultant-final")
        tokenizer.save_pretrained("./models/ideaeden-consultant-final")
        
        return {
            "training_loss": training_result.training_loss,
            "model_path": "./models/ideaeden-consultant-final",
            "training_samples": len(train_data),
            "validation_samples": len(val_data)
        }
```
